// ── Partial Dispatch Modal Logic ── let _dispatchSaleItems = []; let _dispatchSaleId = null; const partialDispatchModalEl = document.getElementById('partialDispatchModal'); // ── Unified open-modal trigger ── document.querySelectorAll('.open-dispatch-modal').forEach(function(btn) { btn.addEventListener('click', function(e) { e.preventDefault(); e.stopPropagation(); const saleId = this.dataset.saleId; const saleNumber = this.dataset.saleNumber; fetch('/sales/api/' + saleId + '/detail/') .then(res => { if (!res.ok) throw new Error('Failed to load sale details'); return res.json(); }) .then(data => { _dispatchSaleId = saleId; document.getElementById('dispatchModalSaleNumber').textContent = '#' + (saleNumber || ''); populateDispatchModal(data.items, data.allow_partial_dispatch !== false); populateDispatchHistory(data.dispatch_history || []); if (partialDispatchModalEl) { const modalInstance = bootstrap.Modal.getInstance(partialDispatchModalEl) || new bootstrap.Modal(partialDispatchModalEl); modalInstance.show(); } }) .catch(err => { console.error(err); alert('Failed to load items for dispatch.'); }); }); }); // ── Sale Detail modal dispatch button ── const saleDetailDispatchBtn = document.getElementById('saleDetailDispatchBtn'); if (saleDetailDispatchBtn && partialDispatchModalEl) { saleDetailDispatchBtn.addEventListener('click', function() { _dispatchSaleId = this.dataset.saleId; document.getElementById('dispatchModalSaleNumber').textContent = '#' + (this.dataset.saleNumber || ''); const allowPartial = window._allowPartialDispatch !== false; populateDispatchModal(_dispatchSaleItems, allowPartial); populateDispatchHistory(window._dispatchHistory || []); const saleDetailModal = document.getElementById('saleDetailModal'); if (saleDetailModal) { const sdInstance = bootstrap.Modal.getInstance(saleDetailModal); if (sdInstance) sdInstance.hide(); } const modalInstance = bootstrap.Modal.getInstance(partialDispatchModalEl) || new bootstrap.Modal(partialDispatchModalEl); modalInstance.show(); }); } function populateDispatchModal(items, allowPartial) { // allowPartial defaults to true if not explicitly false if (allowPartial === undefined) allowPartial = true; const tbody = document.getElementById('dispatchItemsBody'); const toggleContainer = document.getElementById('showDispatchedToggleContainer'); const toggle = document.getElementById('showDispatchedToggle'); const partialBtn = document.getElementById('partialDispatchSubmitBtn'); const dispatchNowHeader = document.getElementById('dispatchNowColHeader'); const infoText = document.getElementById('dispatchInfoText'); if (!tbody) return; // ── Adapt UI based on shop setting ── if (partialBtn) partialBtn.style.display = allowPartial ? '' : 'none'; if (dispatchNowHeader) dispatchNowHeader.style.display = allowPartial ? '' : 'none'; if (infoText) { infoText.innerHTML = allowPartial ? 'Review the items below. Click Full Dispatch to release all remaining items, or adjust quantities and click Partial Dispatch to release only some items now.' : 'This shop requires full dispatch only. Click Full Dispatch to release all remaining items at once.'; } const hasFullyDispatched = items.some(i => i.is_fully_dispatched); if (hasFullyDispatched) { toggleContainer.style.removeProperty('display'); } else { toggleContainer.style.setProperty('display', 'none', 'important'); if (toggle) toggle.checked = false; } function renderRows(showDispatched) { tbody.innerHTML = ''; items.forEach(function(item) { const remaining = parseFloat(item.remaining_quantity); const ordered = parseFloat(item.quantity); const dispatched = parseFloat(item.dispatched_quantity); const done = item.is_fully_dispatched; if (done && !showDispatched) return; const tr = document.createElement('tr'); if (done) tr.classList.add('table-secondary', 'text-muted'); // Dispatch Now cell: only show input if partial dispatch is allowed AND item not done let dispatchNowCell; if (done) { dispatchNowCell = 'Dispatched'; } else if (allowPartial) { dispatchNowCell = ``; } else { // Full dispatch only — show remaining as static text, hidden input carries the value dispatchNowCell = `${remaining} `; } tr.innerHTML = ` ${item.product_name}
${item.sku} ${ordered} ${dispatched} ${done ? 'Done' : remaining} ${dispatchNowCell}`; tbody.appendChild(tr); }); } renderRows(toggle ? toggle.checked : false); if (toggle) { toggle.onchange = () => renderRows(toggle.checked); } } function populateDispatchHistory(history) { const section = document.getElementById('dispatchHistorySection'); const body = document.getElementById('dispatchHistoryBody'); if (!section || !body) return; if (!history || history.length === 0) { section.style.display = 'none'; body.innerHTML = ''; return; } section.style.display = ''; body.innerHTML = ''; history.forEach(function(event, idx) { const itemRows = event.items.map(i => ` ${i.product_name} ${i.qty} ` ).join(''); const card = document.createElement('div'); card.className = 'border rounded-3 mb-2 overflow-hidden'; card.innerHTML = `
Dispatch ${idx + 1} — ${event.dispatched_by}
${event.dispatched_at}
${itemRows}
Product Qty Released
`; body.appendChild(card); }); } // ── Dispatch helper ── function submitDispatch(fillAll) { const inputs = document.querySelectorAll('#dispatchItemsBody .dispatch-qty-input'); const dispatchItems = []; let hasQty = false; inputs.forEach(function(inp) { const max = parseFloat(inp.dataset.max); const qty = fillAll ? max : (parseFloat(inp.value) || 0); if (qty > 0) { hasQty = true; dispatchItems.push({ sale_item_id: parseInt(inp.dataset.itemId), qty: qty }); } }); if (!hasQty) { alert('No remaining quantities to dispatch.'); return; } const activeId = fillAll ? 'fullDispatchBtn' : 'partialDispatchSubmitBtn'; const otherId = fillAll ? 'partialDispatchSubmitBtn' : 'fullDispatchBtn'; const activeBtn = document.getElementById(activeId); const otherBtn = document.getElementById(otherId); const originalHtml = activeBtn ? activeBtn.innerHTML : ''; if (activeBtn) { activeBtn.disabled = true; activeBtn.innerHTML = 'Dispatching...'; } if (otherBtn) otherBtn.disabled = true; const csrfEl = document.querySelector('[name=csrfmiddlewaretoken]'); const csrfToken = csrfEl ? csrfEl.value : ''; fetch('/sales/api/sales/' + _dispatchSaleId + '/dispatch/', { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-CSRFToken': csrfToken }, body: JSON.stringify({ items: dispatchItems }) }) .then(r => r.ok ? r.json() : r.json().then(e => { throw new Error(e.error || 'Dispatch failed'); })) .then(function(res) { if (partialDispatchModalEl) { const modalInstance = bootstrap.Modal.getInstance(partialDispatchModalEl); if (modalInstance) modalInstance.hide(); } if (res.waybill_url) { window.open(res.waybill_url, '_blank'); } alert(res.message || 'Dispatch recorded.'); location.reload(); }) .catch(function(err) { alert(err.message); if (activeBtn) { activeBtn.disabled = false; activeBtn.innerHTML = originalHtml; } if (otherBtn) otherBtn.disabled = false; }); } // ── Full Dispatch: fills all remaining quantities, then submits ── const fullDispatchBtn = document.getElementById('fullDispatchBtn'); if (fullDispatchBtn) { fullDispatchBtn.addEventListener('click', function() { submitDispatch(true); }); } // ── Partial Dispatch: submits only entered quantities ── const partialDispatchSubmitBtn = document.getElementById('partialDispatchSubmitBtn'); if (partialDispatchSubmitBtn) { partialDispatchSubmitBtn.addEventListener('click', function() { submitDispatch(false); }); }
